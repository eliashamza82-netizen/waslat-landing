# Webchat Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a custom live chat to waslat.io where visitors chat from a branded widget and the Waslat team replies from an inbox inside app.waslat.io, all on the existing Firestore.

**Architecture:** A vanilla-JS widget on the static marketing site (`waslat-landing`) signs visitors in with Firebase **anonymous auth** and reads/writes their own conversation directly in Firestore. The team reads/replies from a new protected route in the React app (`ecolor-fixed`). Realtime is Firestore `onSnapshot`; security is enforced entirely by Firestore rules.

**Tech Stack:** Firebase Web SDK v10 (CDN ESM) on the static site; React 19 + react-router 7 + Firebase Firestore in the app; `@firebase/rules-unit-testing` + `tsx` + Firebase emulator for rules tests.

```ts
import { readFileSync } from 'node:fs';
import assert from 'node:assert';
import {
  initializeTestEnvironment,
  assertSucceeds,
  assertFails,
  type RulesTestEnvironment,
} from '@firebase/rules-unit-testing';
import { doc, getDoc, setDoc, collection, addDoc } from 'firebase/firestore';

const PROJECT_ID = 'waslat-rules-test';

function visitor(env: RulesTestEnvironment, uid: string) {
  // Override the decoded token so sign_in_provider === 'anonymous'.
  return env.authenticatedContext(uid, { firebase: { sign_in_provider: 'anonymous' } } as any).firestore();
}
function agent(env: RulesTestEnvironment, uid: string) {
  // Default provider ('custom') is non-anonymous => passes isAuth().
  return env.authenticatedContext(uid).firestore();
}

async function main() {
  const env = await initializeTestEnvironment({
    projectId: PROJECT_ID,
    firestore: { rules: readFileSync('firestore.rules', 'utf8') },
  });
  await env.clearFirestore();

  // Seed: agent 'agent1' has role admin; visitorA owns conv 'cA'.
  await env.withSecurityRulesDisabled(async (ctx) => {
    const db = ctx.firestore();
    await setDoc(doc(db, 'users', 'agent1'), { role: 'admin' });
    await setDoc(doc(db, 'webchatConversations', 'cA'), {
      visitorUid: 'visitorA', name: 'A', email: 'a@x.com', lang: 'en',
      status: 'open', unreadForAgent: true,
    });
    await setDoc(doc(db, 'bookingDays', 'lock1'), { subAccountId: 's1' }); // regression target
  });

  // Visitor can create their OWN conversation.
  await assertSucceeds(setDoc(doc(visitor(env, 'visitorB'), 'webchatConversations', 'cB'), {
    visitorUid: 'visitorB', name: 'B', lang: 'en', status: 'open', unreadForAgent: true,
  }));

  // Visitor CANNOT create a conversation owned by someone else.
  await assertFails(setDoc(doc(visitor(env, 'evil'), 'webchatConversations', 'cE'), {
    visitorUid: 'visitorA', name: 'E', lang: 'en', status: 'open', unreadForAgent: true,
  }));

  // Visitor reads OWN conversation, not others'.
  await assertSucceeds(getDoc(doc(visitor(env, 'visitorA'), 'webchatConversations', 'cA')));
  await assertFails(getDoc(doc(visitor(env, 'visitorB'), 'webchatConversations', 'cA')));

  // Visitor posts a 'visitor' message to own conv; cannot post 'agent' message.
  await assertSucceeds(addDoc(collection(visitor(env, 'visitorA'), 'webchatConversations/cA/messages'), {
    sender: 'visitor', text: 'hi', createdAt: Date.now(),
  }));
  await assertFails(addDoc(collection(visitor(env, 'visitorA'), 'webchatConversations/cA/messages'), {
    sender: 'agent', text: 'spoof', createdAt: Date.now(),
  }));

  // Agent reads any conversation and posts an 'agent' message.
  await assertSucceeds(getDoc(doc(agent(env, 'agent1'), 'webchatConversations', 'cA')));
  await assertSucceeds(addDoc(collection(agent(env, 'agent1'), 'webchatConversations/cA/messages'), {
    sender: 'agent', text: 'hello', createdAt: Date.now(), agentUid: 'agent1',
  }));

  // SECURITY REGRESSION: an anonymous visitor must NOT read tenant data gated only by isAuth().
  await assertFails(getDoc(doc(visitor(env, 'visitorA'), 'bookingDays', 'lock1')));

  await env.cleanup();
  console.log('ALL RULES TESTS PASSED');
}

main().catch((e) => { console.error(e); process.exit(1); });
```

    match /webchatConversations/{convId} {
      allow create: if isVisitor()
        && request.resource.data.visitorUid == request.auth.uid
        && request.resource.data.status == 'open';
      allow read: if isWebchatAgent()
        || (isVisitor() && resource.data.visitorUid == request.auth.uid);

      allow update: if isWebchatAgent()

            && request.resource.data.sender == 'visitor'
            && get(/databases/$(database)/documents/webchatConversations/$(convId)).data.visitorUid == request.auth.uid)
          || (isWebchatAgent() && request.resource.data.sender == 'agent');
        allow update, delete: if isWebchatAgent();
      }
    }

Create `ecolor-fixed/src/pages/LiveChatInbox.tsx`:

```tsx
import { useEffect, useRef, useState } from 'react';
import {
  collection, query, orderBy, onSnapshot, addDoc, doc, updateDoc, serverTimestamp,
} from 'firebase/firestore';
import { db } from '../firebase';
import { useAuth } from '../AuthContext';

interface Conversation {
  id: string;
  name?: string; email?: string; phone?: string; lang?: string;
  status?: string; unreadForAgent?: boolean; lastMessagePreview?: string;
}
interface Message { id: string; sender: 'visitor' | 'agent'; text: string }

export default function LiveChatInbox() {
  const { user } = useAuth();
  const [convs, setConvs] = useState<Conversation[]>([]);
  const [activeId, setActiveId] = useState<string | null>(null);
  const [messages, setMessages] = useState<Message[]>([]);
  const [draft, setDraft] = useState('');
  const prevUnread = useRef(0);
  const endRef = useRef<HTMLDivElement>(null);

  // Conversation list (newest activity first).
  useEffect(() => {
    const q = query(collection(db, 'webchatConversations'), orderBy('lastMessageAt', 'desc'));
    return onSnapshot(q, (snap) => {
      const list = snap.docs.map((d) => ({ id: d.id, ...(d.data() as Omit<Conversation, 'id'>) }));
      const unread = list.filter((c) => c.unreadForAgent).length;
      if (unread > prevUnread.current) new Audio('/notify.mp3').play().catch(() => {});
      prevUnread.current = unread;
      setConvs(list);
    });
  }, []);

  // Active thread + clear unread on open.
  useEffect(() => {
    if (!activeId) return;
    updateDoc(doc(db, 'webchatConversations', activeId), { unreadForAgent: false }).catch(() => {});
    const q = query(collection(db, 'webchatConversations', activeId, 'messages'), orderBy('createdAt', 'asc'));
    return onSnapshot(q, (snap) => {
      setMessages(snap.docs.map((d) => ({ id: d.id, ...(d.data() as Omit<Message, 'id'>) })));
      endRef.current?.scrollIntoView({ behavior: 'smooth' });
    });
  }, [activeId]);

  async function send() {
    const text = draft.trim();
    if (!text || !activeId) return;
    setDraft('');
    await addDoc(collection(db, 'webchatConversations', activeId, 'messages'), {
      sender: 'agent', text, createdAt: serverTimestamp(), agentUid: user?.uid ?? null,
    });
    await updateDoc(doc(db, 'webchatConversations', activeId), {
      lastMessageAt: serverTimestamp(), lastMessagePreview: text, unreadForAgent: false,
    });
  }

  const active = convs.find((c) => c.id === activeId);
  return (
    <div style={{ display: 'flex', height: 'calc(100vh - 64px)' }}>
      <aside style={{ width: 300, borderRight: '1px solid #e5e7eb', overflowY: 'auto' }}>
        <h2 style={{ padding: '12px 16px', margin: 0, fontSize: 16 }}>Live chat</h2>
        {convs.map((c) => (
          <button key={c.id} onClick={() => setActiveId(c.id)}
            style={{ display: 'block', width: '100%', textAlign: 'left', padding: '10px 16px',
              background: c.id === activeId ? '#f1f5f9' : 'transparent', border: 'none',
              borderBottom: '1px solid #f1f5f9', cursor: 'pointer', fontWeight: c.unreadForAgent ? 700 : 400 }}>
            <div>{c.name || 'Visitor'} {c.unreadForAgent ? '•' : ''}</div>
            <div style={{ fontSize: 12, color: '#64748b' }}>{c.email || c.phone || ''}</div>
            <div style={{ fontSize: 12, color: '#94a3b8' }}>{c.lastMessagePreview || ''}</div>
          </button>
        ))}
        {convs.length === 0 && <p style={{ padding: 16, color: '#94a3b8' }}>No conversations yet.</p>}
      </aside>
      <section style={{ flex: 1, display: 'flex', flexDirection: 'column' }}>
        {active ? (
          <>
            <header style={{ padding: 12, borderBottom: '1px solid #e5e7eb' }}>
              <strong>{active.name || 'Visitor'}</strong> · {active.email || active.phone || ''}
            </header>
            <div style={{ flex: 1, overflowY: 'auto', padding: 16 }}>
              {messages.map((m) => (
                <div key={m.id} style={{ textAlign: m.sender === 'agent' ? 'right' : 'left', margin: '6px 0' }}>
                  <span style={{ display: 'inline-block', padding: '8px 12px', borderRadius: 12,
                    background: m.sender === 'agent' ? '#7DD3FC' : '#f1f5f9' }}>{m.text}</span>
                </div>
              ))}
              <div ref={endRef} />
            </div>
            <div style={{ display: 'flex', gap: 8, padding: 12, borderTop: '1px solid #e5e7eb' }}>
              <input value={draft} onChange={(e) => setDraft(e.target.value)}
                onKeyDown={(e) => e.key === 'Enter' && send()} placeholder="Type a reply…"
                style={{ flex: 1, padding: 10, border: '1px solid #cbd5e1', borderRadius: 8 }} />
              <button onClick={send} style={{ padding: '10px 18px', background: '#050508', color: '#fff', border: 'none', borderRadius: 8 }}>Send</button>
            </div>
          </>
        ) : <p style={{ margin: 'auto', color: '#94a3b8' }}>Select a conversation</p>}
      </section>
    </div>
  );
}
```

    body.querySelectorAll('.wc-msg').forEach((n) => n.remove());
    snap.docs.forEach((d) => {
      const m = d.data();
      const el = document.createElement('div');
      el.className = 'wc-msg ' + (m.sender === 'agent' ? 'agent' : 'visitor');
      el.textContent = m.text;
      body.appendChild(el);
    });
    body.scrollTop = body.scrollHeight;
  });
}

async function sendMsg() {
  const input = $('#wc-input');
  const text = input.value.trim();
  if (!text || !convId) return;
  input.value = '';
  await addDoc(collection(db, 'webchatConversations', convId, 'messages'), {
    sender: 'visitor', text, createdAt: serverTimestamp(),
  });
  await updateDoc(doc(db, 'webchatConversations', convId), {
    lastMessageAt: serverTimestamp(), lastMessagePreview: text, unreadForAgent: true,
  });
}
$('#wc-send').onclick = sendMsg;
$('#wc-input').addEventListener('keydown', (e) => { if (e.key === 'Enter') sendMsg(); });
